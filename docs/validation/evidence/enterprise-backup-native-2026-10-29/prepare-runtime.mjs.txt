import assert from 'node:assert/strict';
import fs from 'node:fs';
import path from 'node:path';
import {createHash} from 'node:crypto';
import AdmZip from 'adm-zip';
const base=fs.realpathSync('.private/enterprise-native-20261029');
const source=fs.realpathSync('.superpowers/sdd/2026-10-20-real-postgres-validation/deps/node_modules/@embedded-postgres/windows-x64/native');
const target=path.join(base,'runtime');
const digest=bytes=>createHash('sha256').update(bytes).digest('hex');
const bytes=fs.readFileSync(path.join(base,'vector.v0.8.7-pg17.zip'));
assert.equal(digest(bytes),'acd9d678200e856623f0b7416f6f5d132ef8606a2273ff37f0fdf8a7f2c5d7d2');
const zip=new AdmZip(bytes),entries=zip.getEntries();
assert.ok(entries.length<100);assert.ok(entries.reduce((total,item)=>total+item.header.size,0)<2*1024*1024);
for(const entry of entries)assert.match(entry.entryName,/^(?:include\/|lib\/|share\/|readme\.txt)[a-zA-Z0-9_./-]*$/);
assert.ok(entries.every(entry=>!entry.entryName.split('/').includes('..')));
const dll=zip.readFile('lib/vector.dll');assert.equal(dll.readUInt16LE(0),0x5a4d);const pe=dll.readUInt32LE(0x3c);assert.equal(dll.readUInt32LE(pe),0x4550);assert.equal(dll.readUInt16LE(pe+4),0x8664);
const control=zip.readAsText('share/extension/vector.control');assert.match(control,/default_version\s*=\s*'0\.8\.7'/);
assert.equal(fs.existsSync(target),false,'never overwrite an existing runtime');
fs.cpSync(source,target,{recursive:true,dereference:true,errorOnExist:true,force:false});
for(const entry of entries)if(!entry.isDirectory&&(/^(?:lib\/vector\.dll|share\/extension\/vector[^/]+)$/.test(entry.entryName))){const dest=path.join(target,...entry.entryName.split('/'));assert.ok(path.relative(target,dest)&&!path.relative(target,dest).startsWith('..'));fs.mkdirSync(path.dirname(dest),{recursive:true});fs.writeFileSync(dest,entry.getData(),{flag:'wx'});}
const versions=JSON.parse(fs.readFileSync('.superpowers/sdd/2026-10-20-real-postgres-validation/versions.json','utf8'));
for(const file of versions)assert.equal(digest(fs.readFileSync(path.join(target,'bin',file.name))),file.sha256);
fs.writeFileSync(path.join(base,'runtime-verification.json'),JSON.stringify({postgresVersion:'17.10',pgvectorVersion:'0.8.7',source:'https://github.com/andreiramani/pgvector_pgsql_windows/releases/tag/0.8.7',distribution:'unofficial Windows build, isolated test runtime only',zipSha256:digest(bytes),dllSha256:digest(dll),postgres:versions,readme:zip.readAsText('readme.txt'),files:entries.filter(entry=>!entry.isDirectory).map(entry=>({path:entry.entryName,bytes:entry.header.size}))},null,2)+'\n');
console.log(JSON.stringify({runtime:path.relative(process.cwd(),target),zipSha256:digest(bytes),dllSha256:digest(dll),architecture:'AMD64',originalRuntimeUnchanged:true}));
