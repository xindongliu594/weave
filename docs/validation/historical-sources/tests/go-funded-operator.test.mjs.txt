test('operator parameters require both source files and the explicitly approved Go route, while retained choice remains explicit', () => {
  const args = ['inspect', '--scenario=normal', `--normal-project-id=${id(1)}`, '--version=5', '--confirmed-evidence=docs/test-evidence/synthetic-confirm-never-written.json',
    '--generation-evidence=docs/test-evidence/synthetic-generation-never-written.json', `--go-model-id=${goId}`, '--retain-uncertain-costs=true'];
  assert.equal(parameters(args).retained, true); assert.equal(parameters(args).goModelId, goId);
  assert.throws(() => parameters(args.filter(arg => !arg.startsWith('--generation-evidence='))));
  assert.throws(() => parameters(args.map(arg => arg.startsWith('--go-model-id=') ? '--go-model-id=' + id(9) : arg)));
  assert.throws(() => parameters([...args, '--base-url=https://external.invalid/']));
  assert.throws(() => parameters([...args, '--version=5']));
  const claim = onePostClaim(); claim(); assert.throws(claim);
});

test('unknown outside-project Go fees remain reserved, require scenario attribution and reduce the original funding allowance', () => {
  const project = { id: id(1), createdAt: '2026-10-06T00:00:00.000Z', policy: { maxBudget: 4 } };
  const other = { id: id(2), createdAt: '2026-10-06T00:00:00.000Z', policy: { maxBudget: 5 } };
  const receipt = { id: id(10), projectId: null, runId: null, amount: .0016958, status: 'uncertain', purpose: 'synthetic Go connection test', history: [] };
  const input = { intent: { authorization: { newProjectLimit: 5, newProjectCount: 2, totalAdditionalLimit: 10 }, recordedAt: '2026-10-05T00:00:00.000Z', oldProject: { id: id(99) }, baselineUsageIds: [] },
    ledger: { baselineReceiptIds: [], scenarios: { normal: [receipt.id], recovery: [] } }, projects: [project, other], usage: [], costs: { calls: [receipt] }, configuredProjects: { normal: project.id, recovery: other.id }, scenario: 'normal' };
  const checked = fundedBudgetGuard(input);
  assert.equal(checked.phaseReserved, receipt.amount); assert.equal(checked.allowedProjectCap, 5 - receipt.amount);
  const unassigned = structuredClone(input); unassigned.ledger.scenarios.normal = []; assert.throws(() => fundedBudgetGuard(unassigned));
  const exhausted = structuredClone(input); exhausted.projects[0].policy.maxBudget = 5; assert.throws(() => fundedBudgetGuard(exhausted));
});
