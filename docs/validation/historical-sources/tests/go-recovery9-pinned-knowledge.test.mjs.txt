const oldText=readFileSync(oldPath,'utf8'),newText=readFileSync(newPath,'utf8');
const digest=bytes=>createHash('sha256').update(bytes).digest('hex');
const oldParagraph='本版目标仍是用户批准的真实受控停止/重启/同版检查点恢复：准备好QA且调用有余量后由用户操作员决定停止；若之后授权资料被更新，恢复后QA需再次真实检索/读取当前新版本，引用真实ID/version/SHA，不把旧知识当已刷新。当前两知识样本仍v1，不声称已更新v2或完成恢复。本候选不自动停服/操作服务/更新资料。';
const newParagraph='本版目标仍是用户批准的真实受控停止/重启/同版检查点恢复：准备好QA且调用有余量后由用户操作员决定停止。parent9批准snapshot的shared及QAprivate文档固定v1及其原documentId/SHA；若之后资料目录catalog更新v2，恢复运行仍必须真实knowledge_search/knowledge_read/cite原v1，核对原ID/version/SHA，不读取或引用目录v2。QA恢复后再次真实检索/读取固定v1并给真实引用；目录v2不能替换父snapshot或扩大员工授权，经理仍只读shared原v1。当前两知识样本仍v1，不声称目录已更新v2或恢复已完成。本候选不自动停服/操作服务/更新资料。';
const sf=ts.createSourceFile(newPath,newText,ts.ScriptTarget.Latest,true,ts.ScriptKind.JS);
function declaration(name){for(const statement of sf.statements)if(ts.isVariableStatement(statement))for(const node of statement.declarationList.declarations)if(node.name.getText(sf)===name)return node.initializer;throw Error('Missing source declaration '+name);}
function renderRevision(evidence,agents){
 const node=declaration('revisionRequest');assert.ok(ts.isTemplateExpression(node));let result=node.head.text;
 for(const span of node.templateSpans){
  const expression=span.expression.getText(sf);let value;
  if(expression==='modelId'){const model=declaration('modelId');assert.ok(ts.isStringLiteral(model));value=model.text;}
  else if(['recoveryTimeoutEvidence.sourceEventCount','recoveryTimeoutEvidence.hostCalls','recoveryTimeoutEvidence.wireCalls'].includes(expression))value=evidence[expression.split('.').at(-1)];
  else{assert.equal(expression,'JSON.stringify(project.plan.agents.map(agent => ({ id: agent.id, knowledgeSpaceIds: agent.knowledgeSpaceIds ?? [] })))');value=JSON.stringify(agents.map(agent=>({id:agent.id,knowledgeSpaceIds:agent.knowledgeSpaceIds??[]})));}
  result+=String(value)+span.literal.text;
 }return result;
}
