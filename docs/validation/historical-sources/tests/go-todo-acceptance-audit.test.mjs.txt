test('native Node spec and TAP counters are equivalent with strict full-count, hash and receipt evidence', () => {
  const spec = '\u001b[32m✔ real CLI scenario (1ms)\u001b[0m\r\nℹ tests 12\r\nℹ suites 0\r\nℹ pass 12\r\nℹ fail 0\r\nℹ cancelled 0\r\nℹ skipped 0\r\nℹ todo 0\r\nℹ duration_ms 2\r\n';
  const expected = { tests: 12, pass: 12, fail: 0, cancelled: 0, skipped: 0, todo: 0 };
  assert.deepEqual(parseNodeTestEvidence(spec), expected);
  assert.deepEqual(parseNodeTestEvidence(spec.replaceAll('ℹ', '#')), expected);
  for (const output of [spec + 'ℹ tests 13\n', spec.replace('ℹ pass 12', 'ℹ pass 11'), spec.replace('ℹ skipped 0', 'ℹ skipped 1'), spec.replace('ℹ todo 0\r\n', ''), spec.replaceAll('ℹ', 'echo ℹ')]) assert.throws(() => parseNodeTestEvidence(output));
  const input = fixture();
  for (const section of [input.normal.final, input.recovery.parent, input.recovery.final]) for (const event of section.archive.events.filter(row => row.type === 'tool/result' && row.data.name === 'terminal_test')) event.data.result.value.output = spec;
