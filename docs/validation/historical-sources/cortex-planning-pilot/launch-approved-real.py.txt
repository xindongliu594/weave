"""Requires explicit approval: read only two runtime config values, kept in broker memory.

The Cortex child receives a separate credential-free environment. Never print configuration.
"""
import pathlib,subprocess,sys
assert sys.argv[1:] in [['--approved-real'], ['--approved-real','--project-requirement']], 'Explicit operator approval required'
mode='real-project' if '--project-requirement' in sys.argv else 'real'
root=pathlib.Path('/home/jj/.local/share/weiwu')
repo=root/'releases/weiwu-native-capabilities-20261018'
base=root/'validation/cortex-planning-20261018'
assert not (base/mode).exists(), 'Do not replay an existing paid trial'
matches=[]
for p in pathlib.Path('/proc').iterdir():
 if not p.name.isdigit():continue
 try:
  if b'scripts/local-api.ts' in (p/'cmdline').read_bytes().split(b'\0') and (p/'cwd').resolve()==repo:matches.append(p)
 except (PermissionError,FileNotFoundError,ProcessLookupError):pass
assert len(matches)==1, 'Expected exactly one live Weiwu API'
env={'PATH':str(root/'runtime/node-v24.19.0-linux-x64/bin')+':/usr/bin:/bin','HOME':str(pathlib.Path.home())}
raw=(matches[0]/'environ').read_bytes()
for item in raw.split(b'\0'):
 key,sep,value=item.partition(b'=')
 if sep and key in [b'DATABASE_URL',b'DATA_DIR']:env[key.decode()]=value.decode()
del raw
assert env.get('DATA_DIR')==str(root/'data/pilot/credentials')
assert env.get('DATABASE_URL','').startswith('postgresql://postgres:postgres@127.0.0.1:')
node=root/'runtime/node-v24.19.0-linux-x64/bin/node'
result=subprocess.run([str(node),'--import',str(repo/'node_modules/tsx/dist/loader.mjs'),str(base/'broker.mjs'),str(repo),str(base),mode],cwd=repo,env=env)
sys.exit(result.returncode)
